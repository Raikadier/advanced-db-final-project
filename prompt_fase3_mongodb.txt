import pyodbc
import pandas as pd

STAGING_CONN_STR = (
    "DRIVER={ODBC Driver 17 for SQL Server};"
    "SERVER=(localdb)\\MSSQLLocalDB;"
    "DATABASE=northwind_staging;"
    "Trusted_Connection=yes;"
    "TrustServerCertificate=yes;"
)

def read_staging_table(table_name: str) -> pd.DataFrame:
    """Lee una tabla de staging y retorna un DataFrame."""
    conn = pyodbc.connect(STAGING_CONN_STR)
    df = pd.read_sql(f"SELECT * FROM {table_name}", conn)
    conn.close()
    return df

CORRECCIÓN REQUERIDA:
  La segmentación debe basarse en el total de ventas reales del cliente.

def build_dim_cliente(customers, orders, details):
    from collections import defaultdict

    # Calcular ventas totales reales por cliente cruzando orders y details
    order_to_customer = {o["OrderID"]: o["CustomerID"] for o in orders}
    order_ultima_fecha = {o["OrderID"]: o.get("OrderDate") for o in orders}

    stats = defaultdict(lambda: {"total_ventas": 0.0, "n_ordenes": 0,
                                  "ultima_compra": datetime(1990, 1, 1)})
    for d in details:
        oid = d["OrderID"]
        cid = order_to_customer.get(oid)
        if not cid:
            continue
        valor = float(d.get("STG_ValorNeto") or 0)
        stats[cid]["total_ventas"] += valor
        stats[cid]["n_ordenes"]    += 1
        fecha = order_ultima_fecha.get(oid)
        if fecha and isinstance(fecha, datetime) and fecha > stats[cid]["ultima_compra"]:
            stats[cid]["ultima_compra"] = fecha

    hoy = max((o.get("OrderDate") for o in orders
               if isinstance(o.get("OrderDate"), datetime)), default=datetime(1998,5,6))

    docs = []
    for c in customers:
        cid = c["CustomerID"]
        s = stats[cid]
        dias_inactivo = (hoy - s["ultima_compra"]).days if s["ultima_compra"] > datetime(1990,1,1) else 9999
        # Segmentación basada en ventas reales
        if dias_inactivo > 400:                  seg = "Inactivo"
        elif s["n_ordenes"] <= 1:                seg = "Nuevo"
        elif s["total_ventas"] > 10000:          seg = "Premium"
        else:                                    seg = "Regular"
        docs.append({
            "cliente_id":       cid,
            "company_name":     c.get("CompanyName"),
            "contact_name":     c.get("ContactName"),
            "contact_title":    c.get("ContactTitle"),
            "address":          c.get("Address"),
            "city":             c.get("City"),
            "region":           c.get("Region"),
            "postal_code":      c.get("PostalCode"),
            "country":          c.get("Country"),
            "phone":            c.get("Phone"),
            "fax":              c.get("Fax"),
            "total_ventas_usd": round(s["total_ventas"], 2),
            "n_ordenes":        s["n_ordenes"],
            "segmento_cliente": seg,
        })
    return docs
